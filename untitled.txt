        *,
        CASE

              OR PREV_LOSS_CAUSE <> LOSS_CAUSE

            ELSE 0
        END AS CHANGE_FLAG
    FROM base

grp AS (
    SELECT
        *,
        SUM(CHANGE_FLAG) OVER (

        ) AS LOSS_CAUSE_GROUP

        LOSS_CAUSE
),

        PNC_SK,
        LOSS_CAUSE,

    FROM periods

def calculate_metrics(actual, predicted):

    actual = np.array(actual)
    predicted = np.array(predicted)

    mae = mean_absolute_error(actual, predicted)

    rmse = np.sqrt(
        mean_squared_error(actual, predicted)
    )

    # Avoid division by zero
    non_zero = actual != 0

    if non_zero.sum() > 0:
        mape = np.mean(
            np.abs(
                (actual[non_zero] - predicted[non_zero])
                / actual[non_zero]
            )
        ) * 100
    else:
        mape = np.nan

    # Symmetric MAPE
    denominator = (
        np.abs(actual) + np.abs(predicted)
    )

    valid = denominator != 0

    if valid.sum() > 0:
        smape = np.mean(
            2 * np.abs(
                actual[valid] - predicted[valid]
            ) / denominator[valid]
        ) * 100
    else:
        smape = np.nan

    r2 = r2_score(actual, predicted)

    return {
        "MAE": mae,
        "RMSE": rmse,
        "MAPE (%)": mape,
        "sMAPE (%)": smape,
        "R2": r2
    }

)

def fnol_analysis(d):
    valid = d["FNOL_EFF_DIFF"].dropna()
    bucketed = pd.cut(valid, bins=FNOL_BINS, labels=FNOL_LABELS)
    counts = bucketed.value_counts().reindex(FNOL_LABELS).fillna(0).astype(int)
    pct = (counts / max(len(valid), 1) * 100).round(1)
    table = pd.DataFrame({"Days Bucket": FNOL_LABELS, "Claim Count": counts.values, "% of Claims": pct.values})
    summary = pd.DataFrame({
        "Metric": ["Min Days", "Median Days", "Average Days", "Max Days", "Claim Count"],
        "Value": [
            round(valid.min(), 1) if len(valid) else 0,
            round(valid.median(), 1) if len(valid) else 0,
            round(valid.mean(), 1) if len(valid) else 0,
            round(valid.max(), 1) if len(valid) else 0,
            len(valid),
        ],
    })
    return table, summary

def cause_days_analysis(d):

        causes = [c for c, _ in chain]

def breakdown_table(d, col, label):
    counts = d[col].value_counts()
    pct = (counts / max(len(d), 1) * 100).round(1)
    return pd.DataFrame({label: counts.index, "Claim Count": counts.values, "% of Claims": pct.values})
def cause_to_segment(d):
    dd = d.copy()
    dd["Cause Changed?"] = np.where(dd["CAUSE_CHANGE_COUNT"] > 0, "Cause Changed (1+)", "No Cause Change")
    dd["Segment Changed?"] = np.where(dd["SEGMENT_TRANFER_COUNT"] > 0, "Segment Changed (1+)", "No Segment Change")
    ct = pd.crosstab(dd["Cause Changed?"], dd["Segment Changed?"])
    ct_pct_row = ct.div(ct.sum(axis=1), axis=0).mul(100).round(1)
    cause_changed = dd[dd["Cause Changed?"] == "Cause Changed (1+)"]
    trigger_rate = (round((cause_changed["Segment Changed?"] == "Segment Changed (1+)").mean() * 100, 1)
                     if len(cause_changed) else 0.0)
    headline = pd.DataFrame({
        "Metric": ["Claims with a Cause Change", "...of those, % that ALSO had a Segment Change"],
        "Value": [len(cause_changed), trigger_rate],
    })
    return ct, ct_pct_row, headline, trigger_rate

def parse_cause_days(value):
    if pd.isna(value) or str(value).strip() == "":
        return []
    output = []
    for item in str(value).split(">"):
        item = item.strip()
        if ":" not in item:
            continue
        cause, days = item.rsplit(":", 1)
        cause = cause.strip()
Page 5
        try:
            days = float(days.strip())
        except ValueError:
            continue
        if cause:
            output.append((cause, days))
    return output
def cause_features(value):
    parsed = parse_cause_days(value)
